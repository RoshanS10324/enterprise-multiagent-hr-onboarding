"""PROPOSED, NOT APPLIED. Belongs in HR-Onboarding-Common (for example platform/postgrest_table_client.py), never inside UC-04.
A `TableClient` (select / insert / upsert / update) over the existing PostgREST wrapper. Server-side only: the service-role key stays in the
Common settings and is never passed to UC-04. Every call is filtered; `update` returns the number of rows changed, which UC-04 uses as a
compare-and-swap on the version it read."""

import json
import urllib.error
import urllib.parse
import urllib.request


class PostgrestTableClient:
    def __init__(self, base_url: str, service_key: str, schema: str = "public", timeout: float = 20.0):
        self._base, self._key, self._schema, self._timeout = base_url.rstrip("/"), service_key, schema, timeout

    def _call(self, method, path, body=None, prefer=None):
        headers = {"apikey": self._key, "Authorization": f"Bearer {self._key}", "Accept": "application/json",
                   "Content-Type": "application/json", "Accept-Profile": self._schema, "Content-Profile": self._schema}
        if prefer:
            headers["Prefer"] = prefer
        data = json.dumps(body, default=str).encode() if body is not None else None
        request = urllib.request.Request(self._base + path, data=data, method=method, headers=headers)
        try:
            with urllib.request.urlopen(request, timeout=self._timeout) as response:
                raw = response.read()
                return json.loads(raw) if raw else None
        except urllib.error.HTTPError as exc:
            raise RuntimeError(f"database request failed ({exc.code})") from None  # a code only, never the body or the URL

    @staticmethod
    def _eq(filters):
        return "".join(f"&{c}=eq.{urllib.parse.quote(str(v), safe='')}" for c, v in filters.items())

    def select(self, table, *, filters, columns=None, limit=None):
        if not filters:
            raise ValueError("an unfiltered read is not allowed")
        cols = ",".join(columns) if columns else "*"
        return self._call("GET", f"/rest/v1/{table}?select={cols}{self._eq(filters)}" + (f"&limit={limit}" if limit else "")) or []

    def insert(self, table, row):
        self._call("POST", f"/rest/v1/{table}", dict(row), "return=minimal")  # a duplicate primary key raises

    def upsert(self, table, row, *, on_conflict):
        self._call("POST", f"/rest/v1/{table}?on_conflict={','.join(on_conflict)}", dict(row), "resolution=merge-duplicates,return=minimal")

    def update(self, table, *, filters, values):
        if not filters:
            raise ValueError("an unfiltered update is not allowed")
        changed = self._call("PATCH", f"/rest/v1/{table}?select=case_id{self._eq(filters)}", dict(values), "return=representation")
        return len(changed or [])
